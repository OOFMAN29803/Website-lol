                if token not in self.vocab:
                    self.vocab.append(token)

            print("Setting up word-index mappings...")
            self.word2idx = {word: idx for idx, word in enumerate(self.vocab)}
            self.idx2word = {idx: word for word, idx in self.word2idx.items()}

        for src, tgt in data:
            vocab.update(self.tokenizer(src))
            vocab.update(self.tokenizer(tgt))
        return sorted(list(vocab))

class PositionalEncoding(nn.Module):
    def __init__(self, d_model, dropout=0.1, max_len=5000):
        super().__init__()
        self.dropout = nn.Dropout(p=dropout)

        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
        div_term = torch.exp(torch.arange(0, d_model, 2).float() * (-math.log(10000.0) / d_model))
        pe[:, 0::2] = torch.sin(position * div_term)
        pe[:, 1::2] = torch.cos(position * div_term)
        pe = pe.unsqueeze(0).transpose(0, 1)
        self.register_buffer('pe', pe)

    def forward(self, x):
        x = x + self.pe[:x.size(0), :]
        return self.dropout(x)

class TransformerModel(nn.Module):
    def __init__(self, vocab_size, d_model=512, nhead=8, num_encoder_layers=6, num_decoder_layers=6,
                 dim_feedforward=1024, dropout=0.1):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, d_model)
        self.pos_encoder = PositionalEncoding(d_model, dropout)

        self.transformer = nn.Transformer(d_model, nhead, num_encoder_layers, num_decoder_layers, dim_feedforward, dropout)
        self.fc_out = nn.Linear(d_model, vocab_size)
        self.d_model = d_model

        for p in self.parameters():
            if p.dim() > 1:
                nn.init.xavier_uniform_(p)

    def forward(self, src, tgt, src_mask, tgt_mask, src_padding_mask, tgt_padding_mask, memory_key_padding_mask):
        src = self.embedding(src) * math.sqrt(self.d_model)
        tgt = self.embedding(tgt) * math.sqrt(self.d_model)

        src = self.pos_encoder(src)
        tgt = self.pos_encoder(tgt)

        output = self.transformer(src, tgt, src_mask, tgt_mask, None, src_padding_mask, tgt_padding_mask)
        output = self.fc_out(output)
        return output


class TranslationModule(pl.LightningModule):
    def __init__(self, vocab_size, d_model, nhead, num_encoder_layers, num_decoder_layers, dim_feedforward, dropout,
                 learning_rate, weight_decay, accumulation_steps, clip_grad_norm, dataset, train_dataset, val_dataset):
        super().__init__()
        self.model = TransformerModel(vocab_size, d_model, nhead, num_encoder_layers, num_decoder_layers,
                                      dim_feedforward, dropout)
        self.learning_rate = learning_rate
        self.weight_decay = weight_decay
        self.accumulation_steps = accumulation_steps
        self.clip_grad_norm = clip_grad_norm
        self.dataset = dataset
        self.train_dataset = train_dataset
        self.val_dataset = val_dataset
        self.criterion = nn.CrossEntropyLoss(ignore_index=self.dataset.datasets[0].word2idx['<pad>'])
        self.save_hyperparameters(ignore=['dataset', 'train_dataset', 'val_dataset'])

    def forward(self, src, tgt, src_mask, tgt_mask, src_padding_mask, tgt_padding_mask, memory_key_padding_mask):
        return self.model(src, tgt, src_mask, tgt_mask, src_padding_mask, tgt_padding_mask, memory_key_padding_mask)

    def training_step(self, batch, batch_idx):
        src, tgt = batch
        tgt_input = tgt[:, :-1]
        tgt_out = tgt[:, 1:]
        src_mask, tgt_mask, src_padding_mask, tgt_padding_mask = self.create_mask(src, tgt_input)

        logits = self(src, tgt_input, src_mask, tgt_mask, src_padding_mask, tgt_padding_mask, src_padding_mask)
        loss = self.criterion(logits.contiguous().view(-1, logits.size(-1)), tgt_out.contiguous().view(-1))

        self.log("train_loss", loss, on_step=True, on_epoch=True, prog_bar=True, logger=True)
        return {"loss": loss}

    def validation_step(self, batch, batch_idx):
        src, tgt = batch
        tgt_input = tgt[:, :-1]
        tgt_out = tgt[:, 1:]
        src_mask, tgt_mask, src_padding_mask, tgt_padding_mask = self.create_mask(src, tgt_input)
        logits = self(src, tgt_input, src_mask, tgt_mask, src_padding_mask, tgt_padding_mask, src_padding_mask)
        loss = self.criterion(logits.contiguous().view(-1, logits.size(-1)), tgt_out.contiguous().view(-1))

        _, predicted = torch.max(logits, dim=-1)
        predicted = predicted.view(-1)
        tgt_out_flat = tgt_out.view(-1)
        mask = tgt_out_flat != self.dataset.datasets[0].word2idx['<pad>']
        correct = (predicted == tgt_out_flat) & mask

        # Division by zero fix
        accuracy = correct.sum().float() / mask.sum().float() if mask.sum().float() > 0 else torch.tensor(0.0).to(self.device)

        self.log("val_loss", loss, on_step=False, on_epoch=True, prog_bar=True, logger=True, sync_dist=True)
        self.log("val_accuracy", accuracy, on_step=False, on_epoch=True, prog_bar=True, logger=True, sync_dist=True)
        return {"val_loss": loss, "val_accuracy": accuracy}

    def on_validation_epoch_end(self):
        if self.global_rank == 0:
            self.test_conversation()

    def configure_optimizers(self):
        optimizer = optim.AdamW(self.model.parameters(), lr=self.learning_rate, weight_decay=self.weight_decay)
        scheduler = optim.lr_scheduler.ReduceLROnPlateau(optimizer, mode='min', factor=0.1, patience=2, verbose=True)
        return {
            "optimizer": optimizer,
            "lr_scheduler": {
                "scheduler": scheduler,
                "monitor": "val_loss"
            }
        }

    def create_mask(self, src, tgt):
        tgt_seq_len = tgt.shape[1]
        tgt_mask = nn.Transformer.generate_square_subsequent_mask(tgt_seq_len).to(self.device)
        src_mask = None  # If not using a source mask
        src_padding_mask = (src == self.dataset.datasets[0].word2idx['<pad>'])
        tgt_padding_mask = (tgt == self.dataset.datasets[0].word2idx['<pad>'])
        return src_mask, tgt_mask, src_padding_mask, tgt_padding_mask


    def translate(self, src_sentence, max_len=100):
        self.model.eval()
        with torch.no_grad():
            src_tokens = [self.dataset.datasets[0].word2idx['<bos>']] + [
                self.dataset.datasets[0].word2idx.get(token, self.dataset.datasets[0].word2idx['<unk>']) for token in
                self.dataset.datasets[0].tokenizer(src_sentence.lower())] + [self.dataset.datasets[0].word2idx['<eos>']]
            src = torch.LongTensor(src_tokens).unsqueeze(0).to(self.device)
            src_mask = None  # If not using a source mask
            src_key_padding_mask = (src == self.dataset.datasets[0].word2idx['<pad>'])

            memory = self.model.transformer.encoder(self.model.embedding(src) * math.sqrt(self.model.d_model) + self.model.pos_encoder(self.model.embedding(src)), src_mask, src_key_padding_mask)

            tgt_tokens = [self.dataset.datasets[0].word2idx["<bos>"]]
            for _ in range(max_len):
                tgt = torch.LongTensor(tgt_tokens).unsqueeze(0).to(self.device)
                tgt_mask = nn.Transformer.generate_square_subsequent_mask(len(tgt_tokens)).to(tgt.device)

                tgt_key_padding_mask = (tgt == self.dataset.datasets[0].word2idx['<pad>'])

                output = self.model.transformer.decoder(self.model.embedding(tgt) * math.sqrt(self.model.d_model) + self.model.pos_encoder(self.model.embedding(tgt)), memory, tgt_mask, None, None, tgt_key_padding_mask, src_key_padding_mask)

                output = self.model.fc_out(output)
                prob = output[:, -1, :]
                _, next_word = torch.max(prob, dim=1)
                next_word = next_word.item()
                tgt_tokens.append(next_word)

                if next_word == self.dataset.datasets[0].word2idx['<eos>']:
                    break

            translated_sentence = " ".join([self.dataset.datasets[0].idx2word[idx] for idx in tgt_tokens[1:-1]])
            return translated_sentence

    def train_dataloader(self):
        return DataLoader(self.train_dataset, batch_size=BATCH_SIZE, shuffle=True, num_workers=0, # num_workers=0 for debugging during development
                          pin_memory=True, collate_fn=self.collate_fn, worker_init_fn=self.worker_init_fn)

    def val_dataloader(self):
        return DataLoader(self.val_dataset, batch_size=BATCH_SIZE, shuffle=False, num_workers=0, # num_workers=0 for debugging during development
                          pin_memory=True, collate_fn=self.collate_fn, worker_init_fn=self.worker_init_fn)

    def collate_fn(self, batch):
        src_batch, tgt_batch = [], []
        for src_sample, tgt_sample in batch:
            src_batch.append(src_sample)
            tgt_batch.append(tgt_sample)

        src_batch = nn.utils.rnn.pad_sequence(src_batch, padding_value=self.dataset.datasets[0].word2idx['<pad>'],
                                              batch_first=True)
        tgt_batch = nn.utils.rnn.pad_sequence(tgt_batch, padding_value=self.dataset.datasets[0].word2idx['<pad>'],
                                              batch_first=True)
        return src_batch, tgt_batch

    def worker_init_fn(self, worker_id):  # Keep this for debugging worker issues
        def catch_exceptions(fn, *args, **kwargs):
            try:
                return fn(*args, **kwargs)
            except Exception as e:
                print(f"Exception in worker process {worker_id}:")
                traceback.print_exc(file=sys.stderr)
                raise

        self.train_dataset.__getitem__ = catch_exceptions(self.train_dataset.__getitem__)
        self.val_dataset.__getitem__ = catch_exceptions(self.val_dataset.__getitem__)

class SaveModelCallback(pl.callbacks.Callback):
    def on_epoch_end(self, trainer, pl_module):
        if trainer.global_rank == 0:  # Only save on the main process
            epoch = trainer.current_epoch + 1
            model_path = f"model_epoch_{epoch}.pth"
            torch.save(pl_module.model.state_dict(), model_path) # Save the model's state_dict
            print(f"Model saved at {model_path}")




def main():
    tropicalchat_train_path = "train.json"
    tropicalchat_val_path = "valid_freq.json"

    dailydialog_train_dataset = DailyDialogDataset(split='train')
    dailydialog_val_dataset = DailyDialogDataset(split='validation', vocab=dailydialog_train_dataset.vocab)

    tropicalchat_train_dataset = TropicalChatDataset(tropicalchat_train_path, split='train',
                                                     vocab=dailydialog_train_dataset.vocab)
    tropicalchat_val_dataset = TropicalChatDataset(tropicalchat_val_path, split='valid',
                                                   vocab=dailydialog_train_dataset.vocab)

    train_dataset = ConcatDataset([dailydialog_train_dataset, tropicalchat_train_dataset])
    val_dataset = ConcatDataset([dailydialog_val_dataset, tropicalchat_val_dataset])
    dataset = train_dataset
    VOCAB_SIZE = dataset.datasets[0].VOCAB_SIZE

    model = TranslationModule(
        vocab_size=VOCAB_SIZE, d_model=D_MODEL, nhead=NHEAD, num_encoder_layers=NUM_ENCODER_LAYERS,
        num_decoder_layers=NUM_DECODER_LAYERS, dim_feedforward=DIM_FEEDFORWARD, dropout=DROPOUT,
        learning_rate=LEARNING_RATE, weight_decay=WEIGHT_DECAY, accumulation_steps=ACCUMULATION_STEPS,
        clip_grad_norm=CLIP_GRAD_NORM, dataset=dataset, train_dataset=train_dataset, val_dataset=val_dataset
    )

    early_stop_callback = EarlyStopping(monitor="val_loss", patience=3, min_delta=0.005, mode="min")
    save_model_callback = SaveModelCallback()
    logger = TensorBoardLogger("tb_logs", name="dailydialog_model")

    trainer = pl.Trainer(
        max_epochs=NUM_EPOCHS,
        accelerator="gpu" if torch.cuda.is_available() else "cpu",
        devices=2 if torch.cuda.device_count() >= 2 else 1,
        strategy="ddp" if torch.cuda.device_count() >= 2 else None,
        callbacks=[early_stop_callback, save_model_callback],
        logger=logger,
        accumulate_grad_batches=ACCUMULATION_STEPS,
        gradient_clip_val=CLIP_GRAD_NORM,
        precision="16-mixed" if torch.cuda.is_available() else 32,
        log_every_n_steps=50,
        check_val_every_n_epoch=1,
        num_sanity_val_steps=0, # Remember to set to 2 after debugging
        enable_checkpointing=True,
        deterministic=True  # For reproducibility
    )
    trainer.fit(model)

        sample_context = "Hello, how are you?"
        response = model.translate(sample_context)
        print(f"Sample Dialogue:")
        print(f"Context: {sample_context}")
        print(f"Response: {response}")



if __name__ == "__main__":
    main()
